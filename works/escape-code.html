実装コードを見る：機能どうしをつなぐConnecter

コインのPresenterが通知する「取得数／最大数」を購読し、すべて取ったときだけクリア側に扉を開かせます。コインとクリアのクラスは互いを参照せず、つなぎ方をこのクラスにまとめています。

// ClearCoinConnecter.cs
public class ClearCoinConnecter : MonoBehaviour
{
    [SerializeField] private GameClearPresenter _clear;
    [SerializeField] private CoinPresenter _coin;

    private void Awake()
    {
        _coin.GetCoin
            .Where(x => x.num == x.maxNum)
            .Subscribe(x =>
            {
                _clear.GetAllCoin();
                _coin.Init();
            }).AddTo(this);

        _coin.GetCoin
            .Subscribe(x =>
            {
                _clear.GetCoin(x.num, x.maxNum);
            }).AddTo(this);
    }
}

Viewが通知した当たり判定のうち、プレイヤーのものだけを取り出して取得数を通知します。取得するたびに出現位置のリストからランダムに選び、次の場所へ移動させます。

private Subject<(int num, int maxNum)> _subject = new();

private void OnEnable()
{
    int random = UnityEngine.Random.Range(0, _view.AppearPositionsCount);
    _coinAim.SetInitialPosition(_view.MoveRandomPosition(random));

    _getKeySubscription = _view.GetCoin
        .Where(x => x == _playerCollider)
        .Subscribe(_ =>
        {
            _subject.OnNext((num: _view.GetCoinNum, maxNum: _view.AppearPositionsCount));

            int random = UnityEngine
                .Random.Range(0, _view.AppearPositionsCount);
            _coinAim.SetInitialPosition(_view.MoveRandomPosition(random));
        });
}

private void OnDisable()
{
    _getKeySubscription.Dispose();
}

// Player/PlayerMovement.cs より抜粋（クラス内の一部）
public void Movement()
{
    _moveInput = _input.Player.Move.ReadValue<Vector2>();

    _velocity = transform.right * _moveInput.x + transform.forward * _moveInput.y;

    _velocity.Normalize();

    _charaController.Move(_velocity * _speed * Time.deltaTime);
}

// Player/RotationNeck.cs より抜粋（クラス内の一部）
private void OnEnable()
{
    //Cursorを中心に固定して消す
    Cursor.visible = false;
    Cursor.lockState = CursorLockMode.Locked;
    _input = new();
    _input.Enable();
}

private void MouseLook()
{
    _lookVelocity = _input.Cursor.Look.ReadValue<Vector2>();

    float x = _lookVelocity.x * _sensitivity;
    float y = _lookVelocity.y * _sensitivity;

    _rotation -= y;
    _rotation = Mathf.Clamp(_rotation, -90, 90);

    // cameraを動かす
    _camera.transform.localRotation = Quaternion.Euler(_rotation, 0f, 0f);

    // playerの体を動かす
    transform.Rotate(Vector3.up * x);
}
実装コードを見る：全取得時の演出

コインをすべて集めると「Go to Exit!!」を表示し、LitMotionで拡大縮小をくり返します。クリアしたときにCancellationTokenSourceでアニメーションを止めます。

// Clear/GameClearView.cs より抜粋（クラス内の一部）
private CancellationTokenSource _cancelSource = new();
public CancellationTokenSource CancelSource => _cancelSource;

public void GetAllCoin()
{
    _getAllCoinText.text = "Go to Exit!!";
    LMotion.Create(from: _minGetAllCoinTextScale,
        to: _maxGetAllCoinTextScale,
        duration: _getAllCoinTextDuration)
            .WithLoops(-1, LoopType.Yoyo)
            .BindToLocalScaleXY(_getAllCoinText.transform)
            .ToUniTask(cancellationToken: _cancelSource.Token,
            cancelBehavior: CancelBehavior.Cancel);
}
実装コードを見る：フェード付きのシーン遷移

画面を暗くするフェードの時間だけUniTaskで待ってからシーンを読み込み、読み込みが終わったらSubjectに通知してフェードを明けます。FadeControllerはSingletonなので、シーンをまたいでも同じオブジェクトが使われます。

// Fade/FadeController.cs より抜粋（クラス内の一部）
public class FadeController : Singleton<FadeController>
{
    protected override void Awake()
    {
        base.Awake();

        _subject.Subscribe(_ => FadeOut()).AddTo(this);
        _subject.OnNext(Unit.Default);
    }

    public void FadeIn()
    {
        LMotion.Create(from: minFadeValue, to: maxFadeValue, duration)
            .Bind(x => _canvasGroup.alpha = x).AddTo(this);
    }

    public async UniTask RestartFade()
    {
        FadeIn();
        await UniTask.WaitForSeconds(duration);
        await SceneManager.LoadSceneAsync("MainScene");
        _subject.OnNext(Unit.Default);
    }
}
実装コードを見る：Singletonの基底クラス

ジェネリックな基底クラスにして、AudioManagerやFadeControllerが継承するだけでSingletonになるようにしています。2つ目が作られたときは自分を破棄し、1つだけが残ります。

// Singleton/Singleton.cs
public class Singleton<T> : MonoBehaviour where T : Singleton<T>
{
    private static T _instance;
    public static T Instance
    {
        get
        {
            if (_instance == null)
            {
                _instance = FindAnyObjectByType<T>();

                if (_instance == null)
                {
                    var s = new GameObject(typeof(T).Name);
                    _instance = s.AddComponent<T>();
                }
            }
            return _instance;
        }
    }

    protected virtual void Awake()
    {
        if (_instance == null)
        {
            _instance = this as T;
            DontDestroyOnLoad(gameObject);
        }
        else if (_instance != this)
        {
            Destroy(gameObject);
        }
    }
}
