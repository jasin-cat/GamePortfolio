Minesweeper 実装コード

盤面の生成、マスを連鎖して開く処理、床のアニメーション、スタート地点の選択、ジャンプの実装例です。各項目を開くと、スクリプトの抜粋を確認できます。

実装コードを見る：安全なマスを連鎖して開く

着地したマスが爆弾ならゲームオーバーを通知します。そうでなければQueueとHashSetで幅優先探索を行い、周りに爆弾がないマスから隣のマスへ順に開いていきます。周りに爆弾があるマスは、その数の画像を表示してそこで止めます。

// Map/MapView.cs より抜粋（クラス内の一部）
public void OpenBlock(Vector2Int start)
{
    if (_blocks[start.x, start.y].Type is BlockType.bomb)
    {
        // ゲームオーバー
        _onGameOver.OnNext(Unit.Default);
        return;
    }

    Queue<Vector2Int> queue = new();
    HashSet<Vector2Int> visited = new();

    queue.Enqueue(start);
    visited.Add(start);

    while(queue.Count > 0)
    {
        Vector2Int current = queue.Dequeue();
        Block block = _blocks[current.x, current.y];

        if (block.IsOpened) continue;

        Texture texture = null;
        if (block.SurroundBombCount > 0)
        {
            texture = _blockTextures[block.SurroundBombCount - 1];
        }
        block.Open(texture);

        // 周りに爆弾がある場合は止める
        if (block.SurroundBombCount > 0) continue;

        foreach (var neighbor in GetNeighbors(current, _width))
        {
            if (visited.Contains(neighbor)) continue;
            if (_blocks[neighbor.x, neighbor.y].Type is BlockType.bomb) continue;

            visited.Add(neighbor);
            queue.Enqueue(neighbor);
        }
    }
}

private IEnumerable<Vector2Int> GetNeighbors(Vector2Int pos, int width)
{
    for (int dx = -1; dx <= 1; dx++)
    {
        for (int dy = -1; dy <= 1; dy++)
        {
            if (dx is 0 && dy is 0) continue;
            int nx = pos.x + dx;
            int ny = pos.y + dy;

            if (nx < 0 || nx >= width
                || ny < 0 || ny >= width) continue;

            yield return new Vector2Int(nx, ny);
        }
    }
}
実装コードを見る：着地点から広がる床のアニメーション

着地したマスからの距離（縦・横・斜めのうち大きいほう）が同じマスをまとめて動かし、距離が1つ増えるごとに50ミリ秒待ちます。これで着地点を中心に波が広がるように見えます。アニメーション中に次の着地があっても重ならないよう、フラグで止めています。

// Map/MapAnimation.cs より抜粋（クラス内の一部）
public async UniTask AnimationAsync(Vector3 pos)
{
    if (_isAnimating) return;
    _isAnimating = true;

    // （着地したマスの添字 x, y を探す処理は省略）

    int maxDist = Mathf.Max(
        _blocks.GetLength(0),
        _blocks.GetLength(1)
    );

    for (int dist = 0; dist < maxDist; dist++)
    {
        for (int i = 0; i < _blocks.GetLength(0); i++)
        {
            for (int j = 0; j < _blocks.GetLength(1); j++)
            {
                int currentDist = Mathf.Max(
                    Mathf.Abs(i - x),
                    Mathf.Abs(j - y)
                );

                if (currentDist != dist) continue;

                Block target = _blocks[i, j];

                if (target is null) continue;

                target.AnimationAsync(0.5f).Forget();
            }
        }

        await UniTask.Delay(50);
    }

    _isAnimating = false;
}
実装コードを見る：マス1つの上下運動（LitMotion）

LitMotionのシーケンスで、上がって戻る動きを高さ1、1/2、1/3の順に3回つなげます。Rigidbodyの位置をFixedUpdateのタイミングでMovePositionするので、上に乗っているプレイヤーも物理演算で一緒に押し上げられます。

// Map/Block.cs より抜粋（クラス内の一部）
public async UniTask AnimationAsync(float maxUpValue)
{
    await LSequence.Create()
        .Append(LitmotionHandle(maxUpValue, duration: 0.5f))
        .Append(LitmotionHandle(maxUpValue / 2, duration: 0.3f))
        .Append(LitmotionHandle(maxUpValue / 3, duration: 0.2f))
        .Run();
}

private MotionHandle LitmotionHandle(float maxUpValue, float duration)
{
    float start = _rb.position.y;
    float target = _rb.position.y + maxUpValue;
    return LMotion.Create(start, target, duration)
        .WithLoops(2, loopType: LoopType.Yoyo)
        .WithScheduler(MotionScheduler.FixedUpdate)
        .Bind(y =>
        {
            Vector3 pos = _rb.position;
            pos.y = y;
            _rb.MovePosition(pos);
        });
}
実装コードを見る：ミニマップのクリック位置をワールド座標に変換

RawImage上のクリック位置をローカル座標に直し、画像のサイズで割って0〜1に正規化します。それをミニマップ用カメラの画面座標に変換し、ScreenToWorldPointでワールド座標を求めて通知します。

// UI/RenderTextureGetWorldPosition.cs より抜粋（クラス内の一部）
public void OnPointerClick(PointerEventData eventData)
{
    RectTransform rect = _rawImage.rectTransform;

    if (RectTransformUtility
        .ScreenPointToLocalPointInRectangle(
            rect,
            eventData.position,
            eventData.pressEventCamera,
            out Vector2 localPoint))
    {
        // _rawImageのサイズを取得
        Vector2 size = rect.rect.size;
        // サイズをNormalizeしてほしい押したポジションを取得
        Vector2 normalizePosition = new Vector2(
            (localPoint.x - rect.rect.x) / size.x,
            (localPoint.y - rect.rect.y) / size.y
        );

        Vector3 screenPos = new Vector3(
            normalizePosition.x * _riCamera.pixelWidth,
            normalizePosition.y * _riCamera.pixelHeight,
            _targetDistanceZ
        );

        Vector3 worldPos = _riCamera.ScreenToWorldPoint(screenPos);
        _onClickRawImageWorldPos.OnNext(worldPos);
    }
}
実装コードを見る：sinカーブのジャンプと着地の通知

経過時間を0〜1に直し、0〜πのsinで高さを決めるので、毎回同じ高さ・同じ時間で着地します。着地したらSubjectで通知し、GameManagerがその位置のマスを開いて床のアニメーションを始めます。

// Player/PlayerJump.cs より抜粋（クラス内の一部）
public async UniTask Jump(float height, float maxTime)
{
    if (_isJumping)
    {
        return;
    }
    _isJumping = true;

    float startY = _transform.position.y;
    float timer = 0;

    while (timer < maxTime)
    {
        timer += Time.deltaTime;
        // 0 ~ 1
        float t = timer / maxTime;
        // 0　~　π
        float rad = t * Mathf.PI;
        // 0 ~ 1
        float y = Mathf.Sin(rad);
        // startY ~ height + _transform.position.y ~ startY
        float currentY = startY + y * height;

        _transform.position = new Vector3
        (
            _transform.position.x,
            currentY,
            _transform.position.z
        );
        await UniTask.Yield(PlayerLoopTiming.FixedUpdate);
    }

    _transform.position = new Vector3
    (
        _transform.position.x,
        startY,
        _transform.position.z
    );

    _onDoneJumping.OnNext(Unit.Default);
    _isJumping = false;
}

// GameManger.cs より抜粋（クラス内の一部）
private void SetPlayer(Vector3 v)
{
    _player = Instantiate(_playerPrefab, v, Quaternion.identity);
    _player.DoneJumping
        .Subscribe(pos =>
        {
            _mapView.SetCurrentSelectBlock(pos);
            _mapView.anim(pos);
        }).AddTo(this);
}
