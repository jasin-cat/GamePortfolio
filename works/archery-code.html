Archery 実装コード

個人制作で実装した入力・照準・矢の物理挙動・得点判定・照準UIの実装例です。各項目を開くと、スクリプトの抜粋を確認できます。

実装コードを見る：マウス入力の通知

Input SystemのInputActionに左クリックを割り当て、押したときはtrue、離したときはfalseをR3のSubjectで通知します。入力の取得とゲーム側の処理を分けています。

// Input/InputMouseLeftButton.cs より抜粋
public class InputMouseLeftButton
{
    private InputAction _input = new("Click");
    private Subject<bool> _onClick = new();
    public Observable<bool> OnClick => _onClick;

    public InputMouseLeftButton()
    {
        _input.AddBinding("<Mouse>/leftButton");

        _input.performed += Click;
        _input.canceled += Release;
        _input.Enable();
    }

    private void Click(InputAction.CallbackContext context)
    {
        _onClick.OnNext(value: true);
    }

    private void Release(InputAction.CallbackContext context)
    {
        _onClick.OnNext(value: false);
    }
}
実装コードを見る：構えと発射の切り替え

押したときに弓を構え、ズームと照準UIの演出を開始します。離したときはカメラの向きに矢を生成して発射し、実行中の演出をキャンセルしてから、ズームアウトと照準UIのフェードを同時に待ちます。

// CameraController.cs より抜粋（クラス内の一部）
_inputMouseLeftButton?.OnClick
    .Where(v => v is true)
    .Subscribe(_ =>
    {
        if (!_isPlay) return;
        _isPlay = false;
        if (!_clickable) return;
        _rotBow.SetIsFocus(true);
        SetReleaseArrowCameraPos();
        _zoomCamera.ZoomIn(_cts.Token).Forget();
        _focusTarget.Focus(_cts.Token).Forget();
    })
    .AddTo(_inputSubscriptions);

_inputMouseLeftButton?.OnClick
    .Where(v => v is false)
    .Subscribe(async _ =>
    {
        if (_isPlay) return;
        if (!_clickable) return;

        var direction = _camera.transform.forward;
        var arrow = Instantiate(
            _arrow,
            _camera.transform.position,
            Quaternion.LookRotation(direction)
        );
        arrow.InpactForce(direction);

        _clickable = false;
        _rotBow.SetIsFocus(false);
        _cts.Cancel();
        _cts = new();

        List<UniTask> tasks = new()
        {
            _zoomCamera.ZoomOut(_cts.Token),
            _focusTarget.FocusOut(_cts.Token)
        };

        try
        {
            await tasks;
        }
        catch (OperationCanceledException)
        {
            return;
        }

        SetInitCameraPos();
        _clickable = true;
    })
    .AddTo(_inputSubscriptions);
実装コードを見る：照準操作

構えている間だけマウスの移動量を角度に変換します。角度は上限で制限し、SmoothDampで滑らかに追従させることで、照準が急に跳ねないようにしています。

// RotationBow.cs より抜粋（クラス内の一部）
void Update()
{
    if (!_isFocus) return;

    var delta = _inputMouseDelta.ReadMouseDelta();

    _targety += delta.x * _sensitivity;
    _targetx -= delta.y * _sensitivity;

    _targetx = Mathf.Clamp(_targetx, -_maxRot, _maxRot);
    _targety = Mathf.Clamp(_targety, -_maxRot, _maxRot);

    _rotx = Mathf.SmoothDamp(_rotx, _targetx, ref _rotVelocityx, _smoothTime);
    _roty = Mathf.SmoothDamp(_roty, _targety, ref _rotVelocityy, _smoothTime);

    this.transform.localRotation = Quaternion.Euler(
        _rotx,
        _roty,
        this.transform.localRotation.z
    );
}
実装コードを見る：矢の発射と停止

発射時はRigidbodyに初速を与え、重力を含んだ軌道は物理演算に任せます。的に刺さったらKinematicにして、その場で止めます。

// Arrow.cs より抜粋（クラス内の一部）
public void ZeroSpeed()
{
    _rb.isKinematic = true;
    _speed = 0f;
}

public void InpactForce(Vector3 direction)
{
    _rb.isKinematic = false;
    _rb.linearVelocity = direction.normalized * _speed;
}
実装コードを見る：命中判定と得点計算

矢が当たったら的の子オブジェクトにして刺さった状態にし、的の中心からの距離で10〜6点を判定します。命中数はReactivePropertyで監視し、3本目でリザルト画面を表示します。

// Target.cs より抜粋（クラス内の一部）
void Start()
{
    _cts = new();
    _strikeCount
        .Where(count => count == 3)
        .Subscribe(async _ =>
        {
            await UniTask.Yield();
            _endPanel.Display(_cts.Token, _point).Forget();
        }).AddTo(this);
}

void OnCollisionEnter(Collision collision)
{
    if (!collision.gameObject
        .TryGetComponent(out Arrow component)) return;

    _strikeCount.Value++;

    component.ZeroSpeed();
    component.transform.SetParent(this.transform);
    _currentArrowDistance = Vector3.Distance(
        this.transform.position,
        component.transform.position);

    GetPoint();
    _onArrowHitTarget.OnNext(true);
}

private void GetPoint()
{
    if (_currentArrowDistance <= _tepPointDistance)
    {
        _point += 10;
    }
    else if (_currentArrowDistance <= _ninePointDistance)
    {
        _point += 9;
    }
    else if (_currentArrowDistance <= _eightPointDistance)
    {
        _point += 8;
    }
    else if (_currentArrowDistance <= _sevenPointDistance)
    {
        _point += 7;
    }
    else if (_currentArrowDistance <= _sixPointDistance)
    {
        _point += 6;
    }

    _pointTMPro.text = $"{_point} POINT";
}
実装コードを見る：照準UIのシェーダー

画面中央からの距離で、円の内側・縁・外側を塗り分けます。UVのx方向に画面のアスペクト比を掛けることで、WebGLの画面比率でも円がゆがまないようにしています。

// Shader/RadiusShader.shader より抜粋（フラグメントシェーダー）
half4 frag(Varyings IN) : SV_Target
{
    float2 uv = IN.uv;
    uv -= float2(0.5, 0.5);
    float aspect = _ImageSize.x / _ImageSize.y;
    uv.x *= aspect;
    float distanceFromCenter = length(uv);

    if (distanceFromCenter < _Radius)
    {
        return _CenterColor;
    }

    float outLine = 1.0 - smoothstep(
        _Radius,
        _Radius + _OutLineWidth,
        distanceFromCenter
    );

    if (outLine > 0.3)
    {
        return _OutLineColor * outLine * _OutLineIntensity;
    }

    return _Color;
}
実装コードを見る：照準UIの演出

構えるとLitMotionでシェーダーの半径を絞り、狙っている範囲が狭まっていく様子を表現します。発射後は半径を戻しながら色と縁をフェードさせ、待ち終わったら非表示にします。

// FocusTarget.cs より抜粋（クラス内の一部）
public async UniTask Focus(CancellationToken token)
{
    SetMaterialAlfa("_Color", _outColorAlfa);
    SetMaterialAlfa("_OutLineColor", _outLineColorAlta);
    _image.enabled = true;
    await LMotion.Create(_initRaidus, _maxRadius, _focusDuration)
    .Bind(r => _focusMaterial.SetFloat("_Radius", r))
    .ToUniTask(cancellationToken: token);
}

public async UniTask FocusOut(CancellationToken token)
{
    float currentRadius = _focusMaterial.GetFloat("_Radius");

    LMotion.Create(currentRadius, _initRaidus, _focusDuration / 2f)
        .Bind(r => _focusMaterial.SetFloat("_Radius", r))
        .ToUniTask(cancellationToken: token).Forget();

    List<UniTask> tasks = new();
    tasks.Add
    (
        UniTask.Delay(
        delayTimeSpan: TimeSpan.FromSeconds(_focusDuration / 2.5f),
        ignoreTimeScale: false,
        delayTiming: PlayerLoopTiming.Update,
        cancellationToken: token)
    );

    Fade("_Color", _focusDuration / 3f, token).Forget();
    tasks.Add
    (
        Fade("_OutLineColor", _focusDuration / 4f, token)
    );

    await tasks;
    if (_image is null) return;
    _image.enabled = false;
}
