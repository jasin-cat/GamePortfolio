実装コードを見る：敵のオブジェクトプール

EnemyPool.csの抜粋。最初にまとめて生成した敵をStackで管理し、足りないときだけ追加で生成します。

// EnemyPool.cs より抜粋（クラス内の一部）
private void SetUp()
{
    _Pool = new();
    PooledEnemy instance = null;
    for (int i = 0; i < _initCount; i++)
    {
        instance = Instantiate(_pooled, this.transform);
        instance.SetPool(this);
        instance.gameObject.SetActive(false);
        _Pool.Push(instance);
    }
}

public PooledEnemy GetPooeldEnemy()
{
    // プールが空なら新しく生成する
    if (_Pool.Count <= 0)
    {
        var instance = Instantiate(_pooled, this.transform);
        instance.SetPool(this);
        instance.gameObject.SetActive(true);
        return instance;
    }

    var pooled = _Pool.Pop();
    pooled.gameObject.SetActive(true);
    return pooled;
}

public void ReturnPool(PooledEnemy pooled)
{
    pooled.gameObject.SetActive(false);
    _Pool.Push(pooled);
}

実装コードを見る：敵の行動の流れ

上下に揺れる → 予備動作 → 攻撃 → 一定時間後にプールへ戻る、の流れをLitMotionとUniTaskで順番に実行します。

// EnemyMovement.cs より抜粋（クラス内の一部）
private async UniTask AnimationAsync(float actionWaitTime, CancellationToken token)
{
    // 上下に揺れながら近づく
    await LMotion.Create(-1f, 1f, actionWaitTime / 4)
        .WithLoops(4, LoopType.Yoyo)
        .Bind(v =>
        {
            var velocity = _rb.linearVelocity;
            velocity.y = v;
            _rb.linearVelocity = velocity;
        })
        .ToUniTask(token);

    // 攻撃の予備動作（拡大縮小）
    await LMotion.Create(1f, 0.5f, actionWaitTime / 10)
        .WithLoops(2, loopType: LoopType.Yoyo)
        .Bind(s =>
        {
            this.transform.localScale = Vector3.one * s;
        })
        .ToUniTask(token);
    BeEnemyAttackSpeed();

    await UniTask.Delay(
        TimeSpan.FromSeconds(10f),
        cancellationToken: token
    );

    Release();
}

public async UniTask OutLineAnimationAsync(CancellationToken token, float from, float to, float duration)
{
    await LMotion.Create(from, to, duration)
        .Bind(x => SetOutLine(x))
        .ToUniTask(token);
}

private void SetOutLine(float outLine)
{
    if (_materialPropertyBlock is null)
    {
        _materialPropertyBlock = new();
    }

    _sr.GetPropertyBlock(_materialPropertyBlock);
    _materialPropertyBlock.SetFloat("_OutLineBeginRange", outLine);
    _sr.SetPropertyBlock(_materialPropertyBlock);
}
実装コードを見る：シーンをまたぐフェード（Singleton）

DontDestroyOnLoadのSingletonにして、どのシーンからでも同じフェードを呼べるようにしています。フェード中はレイキャストで入力を遮断します。

// Singleton.cs より抜粋（クラス内の一部）
protected virtual void Awake()
{
    if(_instance is null)
    {
        _instance = this as T;
        DontDestroyOnLoad(this.gameObject);
    }
    else if(_instance != this)
    {
        Destroy(this.gameObject);
    }
}

// FadeController.cs より抜粋（クラス内の一部）
public async UniTask FadeIn(float duration, CancellationToken token)
{
    if (duration is not 0)
    {
        _duraiton = duration;
    }

    _image.raycastTarget = true;

    await LMotion.Create(0f, 1f, _duraiton)
        .Bind(x =>
        {
            _material.SetFloat("_Alpha", x);
        }).ToUniTask(token);

    _image.raycastTarget = false;
}
実装コードを見る：ルール画像を使ったフェードシェーダー

FadeShader.shaderの抜粋。ルール画像のアルファ値と進行度を足し合わせ、画像の形に沿って画面が切り替わるようにしています。

// FadeShader.shader より抜粋（フラグメントシェーダー）
fixed4 frag(v2f IN) : SV_Target
{
    fixed4 color = tex2D(_MainTex, IN.texcoord);
    half ruleAlpha = tex2D(_RuleTex, IN.texcoord).a;
    // _Alpha（0〜1）を -1〜1 に変換してルール画像に足す
    half alpha = saturate(ruleAlpha + (_Alpha * 2 - 1));
    return fixed4(
        _Color.r,
        _Color.g,
        _Color.b,
        color.a * alpha
    );
}
実装コードを見る：ランキングの保存と並び替え

クリアタイムと日時を記録し、タイム順に並び替えて上位20件だけを残します。

// RankingSave.cs より抜粋
public static void Save(float time)
{
    var data = RankingDataToJson.Load();

    data.SetData(time);
    data.TopTwentyDataSortData();
    RankingDataToJson.Save(data);
}

// RankingController.cs / RankingData より抜粋（クラス内の一部）
public void TopTwentyDataSortData()
{
    TopTwentyData.Sort((a,b) => a.Time.CompareTo(b.Time));
    while(TopTwentyData.Count > 20)
    {
        TopTwentyData.RemoveAt(TopTwentyData.Count - 1);
    }
}

public void SetRankingElements(RankingData rd)
{
    for(int i = 0; i < rd.TopTwentyData.Count; i++)
    {
        var element = _topTenRankingElements[i];
        var data = rd.TopTwentyData[i];
        bool isLastPerson =
            rd.LastTimePerson.Time == data.Time
            && rd.LastTimePerson.Date == data.Date;

        element.Initialize(
            data.Time,
            data.Date,
            isLastPerson,
            i + 1);
    }
}
