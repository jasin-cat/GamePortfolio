実装コードを見る：状態遷移

StateMachine.csの抜粋。各状態の終了処理からUniRxで通知し、次の状態を開始する仕組みです。

// StateMachine.cs より抜粋（クラス内の一部）
// Init内：終了通知を受け取って次の状態を開始
// Exitが終わったらEnterを呼ぶ
_onCompletedExitDisposable = _onCompletedExit
    .Subscribe(_ =>
    {
        _currentState.Enter();
    });

public void ChangeState(State state)
{
    if(_states.TryGetValue(state, out IState currentState))
    {
        IState oldState = _currentState;
        _currentState = currentState;
        _currentStateType = state;

        oldState.Exit();
    }
}

// 各状態の終了処理から呼び出す
public void OnCompletedExitChangeValue()
{
    _onCompletedExit.OnNext(Unit.Default);
}
実装コードを見る：カード選択・ホールド

選択枚数の上限を確認して手札と場を移動し、ホールドしたカードを場に戻します。

// CardListModel.cs より抜粋（クラス内の一部）
public void HandToPlay(GameObject card, List<CardModel> play, int maxPlay)
{
    if(!_cardModelDictionary
        .TryGetValue(
            key: card,
            value: out CardModel model)) return;

    if (_play.Contains(model))
    {
        RemovePlay(model);
        AddHand(model);
    }
    else if(play.Count < maxPlay)
    {
        RemoveHand(model);
        AddPlay(model);
    }
}

public CardModel HoldToPlay(int maxPlay)
{
    if(_play.Count == maxPlay || _hold is null) return null;

    var hold = _hold;
    AddPlay(_hold);
    RemoveHold();
    return hold;
}

実装コードを見る：ダメージ・回復計算

回復を反映し、相手の攻撃と自分の防御からダメージを求めてHPを更新します。

// CalculationButtle.cs より抜粋（クラス内の一部）
public PlayerModel Calculation(PlayerModel player, PlayerModel other)
{
    PlayerModel model = player;
    _oldHP = player.HP;
    model.SetOldHP(_oldHP);
    model.AddHP(model.Recovery);

    _damage = player.Defense - other.Attack;

    if(_damage > 0)
    {
        _damage = 0;
    }
    else
    {
        _damage *= -1;
    }

    _mitigation = model.Defense;

    model.SetDamage(_damage);
    _damage *= -1;
    model.AddHP(_damage);
    model.SetMitigation(_mitigation);
    return model;
}

// CursorInputHandler.cs より抜粋（クラス内の一部）
private void OnMove(InputAction.CallbackContext context)
{
    if(_device == context.control.device || _device == null)
    {
        _stickDirection.Value = context.ReadValue<Vector2>();
    }
}

private void OnStop(InputAction.CallbackContext context)
{
    if(_device == context.control.device || _device == null)
    {
        _stickDirection.Value = Vector2.zero;
    }
}

実装コードを見る：アセットの非同期読み込み

AddressablesとUniTaskでアセットを読み込み、キャンセル時の解放も設定しています。

// LoadAseet.cs より抜粋（クラス内の一部）
public static async UniTask<T> GetAsset<T>(string path, CancellationToken token)
{
    return await Addressables.LoadAssetAsync<T>(path)
        .ToUniTask(
            cancellationToken: token,
            autoReleaseWhenCanceled: true // これをtrueにするとキャンセル時にAddressable解放処理を行う
        );
}
